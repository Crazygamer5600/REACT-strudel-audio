import './App.css';
import { useEffect, useRef, useState } from "react";
import { StrudelMirror } from '@strudel/codemirror';
import { evalScope } from '@strudel/core';
import { drawPianoroll } from '@strudel/draw';
import { initAudioOnFirstClick } from '@strudel/webaudio';
import { transpiler } from '@strudel/transpiler';
import { getAudioContext, webaudioOutput, registerSynthSounds } from '@strudel/webaudio';
import { registerSoundfonts } from '@strudel/soundfonts';
import { stranger_tune } from './tunes';
import console_monkey_patch, { getD3Data } from './console-monkey-patch';

export default function StrudelDemo() {
  const [procText, setProcText] = useState(stranger_tune);
  const [radioValue, setRadioValue] = useState('on');
  const editorContainerRef = useRef(null);
  const canvasRef = useRef(null);
  const globalEditorRef = useRef(null);
  const hasRun = useRef(false);

  // Handle external D3 events
  useEffect(() => {
    const handleD3Data = (event) => console.log(event.detail);
    document.addEventListener("d3Data", handleD3Data);
    console_monkey_patch();
    return () => document.removeEventListener("d3Data", handleD3Data);
  }, []);

  // Initialize StrudelMirror
  useEffect(() => {
    if (hasRun.current) return;
    hasRun.current = true;

    const canvas = canvasRef.current;
    const ctx = canvas.getContext('2d');
    canvas.width *= 2;
    canvas.height *= 2;

    const drawTime = [-2, 2];

    const globalEditor = new StrudelMirror({
      defaultOutput: webaudioOutput,
      getTime: () => getAudioContext().currentTime,
      transpiler,
      root: editorContainerRef.current,
      drawTime,
      onDraw: (haps, time) => drawPianoroll({ haps, time, ctx, drawTime, fold: 0 }),
      prebake: async () => {
        initAudioOnFirstClick();
        const loadModules = evalScope(
          import('@strudel/core'),
          import('@strudel/draw'),
          import('@strudel/mini'),
          import('@strudel/tonal'),
          import('@strudel/webaudio'),
        );
        await Promise.all([loadModules, registerSynthSounds(), registerSoundfonts()]);
      },
    });

    globalEditorRef.current = globalEditor;
    globalEditor.setCode(processText(procText, radioValue));
  }, [procText, radioValue]);

  // --- Logic functions ---
  const processText = (text, mode) => {
    const replace = mode === 'hush' ? '_' : '';
    return text.replaceAll('<p1_Radio>', replace);
  };

  const handleProcess = () => {
    const newText = processText(procText, radioValue);
    globalEditorRef.current?.setCode(newText);
  };

  const handlePlay = () => {
    globalEditorRef.current?.evaluate();
  };

  const handleProcAndPlay = () => {
    if (globalEditorRef.current?.repl?.state?.started) {
      handleProcess();
      handlePlay();
    }
  };

  const handleStop = () => {
    globalEditorRef.current?.stop();
  };

  // --- UI ---
  return (
    <div>
      <h2>Strudel Demo</h2>
      <main>
        <div className="container-fluid">
          <div className="row">
            {/* Left: Text area */}
            <div className="col-md-8" style={{ maxHeight: '50vh', overflowY: 'auto' }}>
              <label htmlFor="proc" className="form-label">
                Text to preprocess:
              </label>
              <textarea
                id="proc"
                className="form-control"
                rows="15"
                value={procText}
                onChange={(e) => setProcText(e.target.value)}
              />
            </div>

            {/* Right: Buttons */}
            <div className="col-md-4">
              <nav>
                <button onClick={handleProcess} className="btn btn-outline-primary">Preprocess</button>
                <button onClick={handleProcAndPlay} className="btn btn-outline-primary">Proc & Play</button>
                <br />
                <button onClick={handlePlay} className="btn btn-outline-primary">Play</button>
                <button onClick={handleStop} className="btn btn-outline-primary">Stop</button>
              </nav>
            </div>
          </div>

          {/* Editor + Controls */}
          <div className="row">
            <div className="col-md-8" style={{ maxHeight: '50vh', overflowY: 'auto' }}>
              <div ref={editorContainerRef} id="editor" />
              <div id="output" />
            </div>

            <div className="col-md-4">
              <div className="form-check">
                <input
                  className="form-check-input"
                  type="radio"
                  name="p1"
                  id="p1_on"
                  value="on"
                  checked={radioValue === 'on'}
                  onChange={(e) => {
                    setRadioValue(e.target.value);
                    handleProcAndPlay();
                  }}
                />
                <label className="form-check-label" htmlFor="p1_on">
                  p1: ON
                </label>
              </div>

              <div className="form-check">
                <input
                  className="form-check-input"
                  type="radio"
                  name="p1"
                  id="p1_hush"
                  value="hush"
                  checked={radioValue === 'hush'}
                  onChange={(e) => {
                    setRadioValue(e.target.value);
                    handleProcAndPlay();
                  }}
                />
                <label className="form-check-label" htmlFor="p1_hush">
                  p1: HUSH
                </label>
              </div>
            </div>
          </div>
        </div>

        <canvas ref={canvasRef} id="roll"></canvas>
      </main>
    </div>
  );
}

import { evalScope } from "@strudel/core";

import { initAudioOnFirstClick, webaudioOutput, getAudioContext, registerSynthSounds } from "@strudel/webaudio";

export function initStrudelEditor({ editorContainer, canvas, procText, radioValue }) {
  const ctx = canvas.getContext("2d");
  canvas.width *= 2;
  canvas.height *= 2;
  const drawTime = [-2, 2];

  const editor = new StrudelMirror({
    defaultOutput: webaudioOutput,
    getTime: () => getAudioContext().currentTime,
    transpiler,
    root: editorContainer,
    drawTime,
    onDraw: (haps, time) => drawPianoroll({ haps, time, ctx, drawTime, fold: 0 }),
    prebake: async () => {

        import("@strudel/tonal"),
        import("@strudel/webaudio")
      );
      await Promise.all([loadModules, registerSynthSounds(), registerSoundfonts()]);
    },
  });

  const initialCode = processText(procText, radioValue);
  editor.setCode(initialCode);

  return editor;
}

export function initStrudelEditor({ editorContainer, canvas, procText, radioValue }) {
  const ctx = canvas.getContext("2d");
  canvas.width *= 2;
  canvas.height *= 2;
  const drawTime = [-2, 2];

  const editor = new StrudelMirror({
    defaultOutput: webaudioOutput,
    getTime: () => getAudioContext().currentTime,
    transpiler,
    root: editorContainer,
    drawTime,
    onDraw: (haps, time) => drawPianoroll({ haps, time, ctx, drawTime, fold: 0 }),
    prebake: async () => {
      initAudioOnFirstClick();

        import("@strudel/draw"),

        import("@strudel/tonal"),
        import("@strudel/webaudio")
      );

// console-monkey-patch.js
// This patches console.log to dispatch a "d3Data" event when certain data is logged.
// It also exports a helper for retrieving the last dispatched data.

let lastD3Data = null;

export default function console_monkey_patch() {
  const originalLog = console.log;

  console.log = function (...args) {
    // Forward all logs normally
    originalLog.apply(console, args);

    // If a special D3-like data object is detected, dispatch it
    const d3Candidate = args.find(
      (arg) => arg && typeof arg === 'object' && arg.type === 'd3Data'
    );

    if (d3Candidate) {
      lastD3Data = d3Candidate;
      const event = new CustomEvent('d3Data', { detail: d3Candidate });
      window.dispatchEvent(event);
    }
  };
}

export function getD3Data() {
  return lastD3Data;
}

The default export is a callable function (console_monkey_patch()), so your existing line

getD3Data remains a named export.

import React, { useEffect, useRef } from "react";
import * as d3 from "d3";

export default function BPMGraph({ bpmTimeline = [], currentTime = 0, width = 800, height = 150, playing }) {
    const svgRef = useRef();

    useEffect(() => {
        if (!playing || !bpmTimeline.length) return; // protect from empty/undefined

        const svg = d3.select(svgRef.current);
        svg.selectAll("*").remove();

        const margin = { top: 10, right: 20, bottom: 20, left: 40 };
        const innerWidth = width - margin.left - margin.right;
        const innerHeight = height - margin.top - margin.bottom;

        const g = svg.append("g").attr("transform", translate(${margin.left},${margin.top}));

        const timeWindow = 5;
        const visibleData = bpmTimeline.filter(d => d.time >= currentTime - timeWindow);

        const xScale = d3.scaleLinear()
            .domain([currentTime - timeWindow, currentTime])
            .range([0, innerWidth]);

        const yScale = d3.scaleLinear()
            .domain([0, 1])
            .range([innerHeight, 0]);

        g.append("g").attr("transform", translate(0,${innerHeight})).call(d3.axisBottom(xScale).ticks(5));
        g.append("g").call(d3.axisLeft(yScale));

        const lineData = visibleData.map(d => {
            const secondsPerBeat = 60 / d.bpm;
            const beatPosition = (d.time % secondsPerBeat) / secondsPerBeat;
            const value = Math.cos(beatPosition * Math.PI * 2) * 0.5 + 0.5; // pulse 0–1
            return { time: d.time, value };
        });

        const line = d3.line()
            .x(d => xScale(d.time))
            .y(d => yScale(d.value))
            .curve(d3.curveMonotoneX);

        g.append("path")
            .datum(lineData)
            .attr("fill", "none")
            .attr("stroke", "#ff6600")
            .attr("stroke-width", 2)
            .attr("d", line);

    }, [bpmTimeline, currentTime, playing, width, height]);

    return React.createElement("svg", { ref: svgRef, width, height });
}
